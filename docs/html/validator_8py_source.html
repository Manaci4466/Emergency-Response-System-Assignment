1"""@brief Phase 7: independent plan validator.
2
3INDEPENDENT of the planner: it reads plan.json and the raw scenario files and uses only
4data_loader and cost_model (the model's DEFINITION of travel time and P(Blocked)). It never imports
5search, allocation or simulator; a test checks that. Travel times are recomputed by summing the edge
6costs along the path given in the plan, so a wrong route or a wrong cost in the planner shows up here.
7
8Errors (the plan breaks a hard constraint):
9 - schema: entry is not {responder_id: str, incident_id: str, path: non-empty list, time: number ≥ 0}
10 - unknown_id: responder or incident id not in the scenario
11 - duplicate: incident assigned more than once (the extra entry is not checked further)
12 - capability: responder lacks the required capability
13 - path_start: first node is not where the responder is at that moment
14 (start_location for its first job, the previous incident's location afterwards)
15 - path_end: last node is not the incident location
16 - no_road: consecutive nodes without a DIRECTED road between them
17 - blocked_road: road with P(Blocked) ≥ tau
18 - repeated_node: the path visits a node twice
19 - before_timestamp: job starts before the incident is known (clairvoyance)
20 - before_available: job starts before the responder's time_until_available
21 - overlap: per responder (sorted by time) a job starts before the previous arrival +
22 time_required (recomputed, tolerance 1e-6)
23
24Warnings: late (recomputed arrival > deadline) and unhandled (incident missing from the plan).
25
26@note LIMITATION: validity is relative to the BELIEVED road states (P(Blocked) < tau) and to EXPECTED
27travel times under the model. A valid plan can still use a road that is in reality blocked; the
28validator shares the cost model, so a wrong model is not detected here (the unit tests do that).
29"""

73def _schema_ok(entry: Any) -> bool:
74 """@brief Whether a plan entry has exactly the four keys with the right types.
75 @param entry One element of plan.json.
76 @return True if valid (bool is rejected as a number).
77 """
78 if not isinstance(entry, dict) or set(entry) != KEYS:
79 return False
80 t = entry["time"]
81 return (
82 isinstance(entry["responder_id"], str)
83 and isinstance(entry["incident_id"], str)
84 and isinstance(entry["path"], list) and len(entry["path"]) > 0
85 and all(isinstance(n, str) for n in entry["path"])
86 and isinstance(t, (int, float)) and not isinstance(t, bool) # bool is a subclass of int
87 and math.isfinite(t) and t >= 0
88 )

91def travel_time(path: list[str], costs: dict[tuple[str, str], EdgeCost]) -> float | None:
92 """@brief Expected travel time = Σ edge costs along the path.
93
94 Uses base · status_mult · traffic_mult also for blocked roads, so the arrival can still be
95 computed; the blocked road itself is reported as a separate error.
96 @param path Node ids of the route.
97 @param costs Edge costs.
98 @return The travel time in seconds, or None if a road does not exist.
99 """
100 total = 0.0
101 for a, b in zip(path, path[1:]):
102 c = costs.get((a, b))
103 if c is None:
104 return None
105 total += c.base * c.status_mult * c.traffic_mult
106 return total

109def validate(
110 plan: Any, scenario: Scenario, tau: float = 0.5, unobserved_traffic: float = 1.0
111) -> tuple[list[Issue], list[Issue]]:
112 """@brief Validate a plan against the hard constraints.
113 @param plan The parsed plan.json.
114 @param scenario The scenario (raw data).
115 @param tau Blocked threshold used for the plan.
116 @param unobserved_traffic Multiplier used for the plan.
117 @return (errors, warnings) as lists of Issue.
118 """
119 errors: list[Issue] = []
120 warnings: list[Issue] = []
121 if not isinstance(plan, list):
122 return [Issue("schema", "plan is not a list")], warnings
123
124 costs = edge_costs(scenario, tau=tau, unobserved_traffic=unobserved_traffic)
125 responders = {r.id: r for r in scenario.responders}
126 incidents = {i.id: i for i in scenario.incidents}
127 seen: set[str] = set()
128 jobs: list[_Job] = []
129
130 # Pass 1: per entry (schema, ids, duplicates, capability, path shape, start time)
131 for k, entry in enumerate(plan):
132 where = f"entry {k}"
133 if not _schema_ok(entry):
134 errors.append(Issue("schema", f"{where}: wrong keys or types: {entry!r}"))
135 continue
136 rid, iid, path, time = entry["responder_id"], entry["incident_id"], entry["path"], entry["time"]
137 where = f"entry {k} ({rid} -> {iid})"
138 if rid not in responders or iid not in incidents:
139 errors.append(Issue("unknown_id", f"{where}: unknown responder or incident id"))
140 continue
141 if iid in seen:
142 errors.append(Issue("duplicate", f"{where}: {iid} is assigned more than once"))
143 continue
144 seen.add(iid)
145 resp, inc = responders[rid], incidents[iid]
146
147 if inc.required_capability not in resp.capabilities:
148 errors.append(Issue("capability", f"{where}: {rid} lacks {inc.required_capability!r}"))
149 if path[-1] != inc.location:
150 errors.append(Issue("path_end", f"{where}: path ends at {path[-1]}, incident is at {inc.location}"))
151 if len(set(path)) != len(path):
152 errors.append(Issue("repeated_node", f"{where}: path visits a node twice: {path}"))
153 for a, b in zip(path, path[1:]):
154 if (a, b) not in costs:
155 errors.append(Issue("no_road", f"{where}: no road {a}->{b}"))
156 elif not costs[(a, b)].usable:
157 errors.append(Issue("blocked_road", f"{where}: road {a}->{b} has "
158 f"P(Blocked) = {costs[(a, b)].p_blocked:.3f} >= tau {tau}"))
159 if time < inc.timestamp:
160 errors.append(Issue("before_timestamp", f"{where}: starts at {time} before the "
161 f"incident is known at {inc.timestamp}"))
162 if time < resp.time_until_available:
163 errors.append(Issue("before_available", f"{where}: starts at {time}, {rid} is "
164 f"busy until {resp.time_until_available}"))
165 jobs.append(_Job(k, rid, inc, path, time, _arrival(time, travel_time(path, costs))))
166
167 # Pass 2: per responder, in time order: path start location and overlap
168 for rid, resp in responders.items():
169 # free_at is about the PREVIOUS job only; the initial busy time is checked in pass 1
170 location, free_at = resp.start_location, None
171 for job in sorted((j for j in jobs if j.responder_id == rid), key=lambda j: (j.time, j.index)):
172 where = f"entry {job.index} ({rid} -> {job.incident.id})"
173 if job.path[0] != location:
174 errors.append(Issue("path_start", f"{where}: path starts at {job.path[0]}, "
175 f"{rid} is at {location}"))
176 if free_at is not None and job.time < free_at - TOLERANCE:
177 errors.append(Issue("overlap", f"{where}: starts at {job.time:.2f}, previous "
178 f"job ends at {free_at:.2f}"))
179 # After the job the responder is at the incident location (decision 8).
180 location = job.incident.location
181 free_at = None if job.arrival is None else job.arrival + job.incident.time_required
182
183 # Warnings: late arrivals and unhandled incidents
184 for job in jobs:
185 if job.arrival is not None and job.arrival > job.incident.deadline:
186 warnings.append(Issue("late", f"{job.incident.id}: arrives {job.arrival:.2f}, "
187 f"deadline {job.incident.deadline} (+{job.arrival - job.incident.deadline:.2f} s)"))
188 for iid in incidents:
189 if iid not in seen:
190 warnings.append(Issue("unhandled", f"{iid} is not in the plan"))
191 return errors, warnings

203def plan_statistics(
204 plan: list[dict], scenario: Scenario, tau: float = 0.5, unobserved_traffic: float = 1.0
205) -> dict[str, float | None]:
206 """@brief The four required statistics recomputed from plan.json alone.
207 @param plan A schema-valid plan.
208 @param scenario The scenario.
209 @param tau Blocked threshold.
210 @param unobserved_traffic Multiplier for roads without observations.
211 @return Dict with handled, responders_used, avg_response_time, deadlines_met_pct.
212 """
213 costs = edge_costs(scenario, tau=tau, unobserved_traffic=unobserved_traffic)
214 incidents = {i.id: i for i in scenario.incidents}
215 arrivals: dict[str, float] = {}
216 for e in plan:
217 travel = travel_time(e["path"], costs)
218 if travel is not None and e["incident_id"] in incidents:
219 arrivals.setdefault(e["incident_id"], e["time"] + travel)
220 response = [arrivals[i] - incidents[i].timestamp for i in arrivals]
221 on_time = sum(1 for i in arrivals if arrivals[i] <= incidents[i].deadline)
222 return {
223 "handled": len(arrivals),
224 "responders_used": len({e["responder_id"] for e in plan}),
225 "avg_response_time": sum(response) / len(response) if response else None,
226 "deadlines_met_pct": 100.0 * on_time / len(incidents) if incidents else 100.0,
227 }
228
229
230_STAT_PATTERNS = {
231 "handled": r"\*\s*(\d+) incidents handled",
232 "responders_used": r"\*\s*(\d+) responders used",
233 "avg_response_time": r"Average response time\s*:\s*([\d.]+|n/a)",
234 "deadlines_met_pct": r"Deadlines met\s*:\s*([\d.]+)%",
235}

238def check_statistics(
239 plan: list[dict], scenario: Scenario, statistics_text: str,
240 tau: float = 0.5, unobserved_traffic: float = 1.0,
241) -> list[Issue]:
242 """@brief Compare the four lines of statistics.txt with values recomputed from plan.json.
243 @param plan A schema-valid plan.
244 @param scenario The scenario.
245 @param statistics_text Contents of statistics.txt.
246 @param tau Blocked threshold.
247 @param unobserved_traffic Multiplier for roads without observations.
248 @return List of "statistics" Issues (empty if everything matches within the printed precision).
249 """
250 expected = plan_statistics(plan, scenario, tau, unobserved_traffic)
251 errors: list[Issue] = []
252 for key, pattern in _STAT_PATTERNS.items():
253 m = re.search(pattern, statistics_text)
254 if m is None:
255 errors.append(Issue("statistics", f"{key}: line not found in statistics.txt"))
256 continue
257 value, want = m.group(1), expected[key]
258 if value == "n/a" or want is None:
259 ok = value == "n/a" and want is None
260 elif key in ("handled", "responders_used"):
261 ok = int(value) == want
262 else: # printed with one decimal
263 ok = abs(float(value) - want) <= 0.05 + 1e-9
264 if not ok:
265 errors.append(Issue("statistics", f"{key}: statistics.txt says {value}, plan gives {want}"))
266 return errors
