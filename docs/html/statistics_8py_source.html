69def compute_statistics(
70 incidents: Sequence[Incident], assignments: Sequence[Assignment], unhandled: Sequence[str]
71) -> Statistics:
72 """@brief Compute the statistics of a plan.
73 @param incidents All incidents of the scenario (the denominator of deadlines met).
74 @param assignments The plan's assignments.
75 @param unhandled Ids of unhandled incidents.
76 @return The Statistics.
77 @exception ValueError If an incident is assigned more than once.
78 """
79 by_id = {i.id: i for i in incidents}
80 arrival = {a.incident_id: a.arrival for a in assignments}
81 if len(arrival) != len(assignments):
82 raise ValueError("an incident was assigned more than once")
83 response = [arrival[i] - by_id[i].timestamp for i in arrival]
84 on_time = {i for i in arrival if arrival[i] <= by_id[i].deadline}
85 late = sorted(((i, arrival[i] - by_id[i].deadline) for i in arrival if i not in on_time),
86 key=lambda x: id_key(x[0]))
87 per_priority = {
88 p: PriorityStats(
89 total=sum(1 for i in incidents if i.priority == p),
90 handled=sum(1 for i in arrival if by_id[i].priority == p),
91 on_time=sum(1 for i in on_time if by_id[i].priority == p),
92 )
93 for p in PRIORITIES
94 if any(i.priority == p for i in incidents)
95 }
96 total = len(incidents)
97 return Statistics(
98 total=total,
99 handled=len(arrival),
100 responders_used=len({a.responder_id for a in assignments}),
101 avg_response_time=sum(response) / len(response) if response else None,
102 on_time=len(on_time),
103 deadlines_met_pct=100.0 * len(on_time) / total if total else 100.0,
104 per_priority=per_priority,
105 late=late,
106 unhandled=sorted(unhandled, key=id_key),
107 weighted_missed=sum(WEIGHTS[by_id[i].priority] for i, _ in late)
108 + sum(WEIGHTS[by_id[i].priority] for i in unhandled),
109 )

112def format_statistics(stats: Statistics, header: str = "", notes: Sequence[str] = ()) -> str:
113 """@brief The text of statistics.txt.
114
115 Starts with the four required lines in the assignment's format, then details and a Definitions
116 block.
117 @param stats The statistics.
118 @param header Settings shown after "Details" (policy, tau, ...).
119 @param notes Extra detail lines from the run (heuristic used, node-limit warnings).
120 @return The complete text.
121 """
122 avg = "n/a" if stats.avg_response_time is None else f"{stats.avg_response_time:.1f} s"
123 lines = [
124 "System statistics:",
125 "",
126 f" * {stats.handled} incidents handled",
127 f" * {stats.responders_used} responders used",
128 f" * Average response time: {avg}",
129 f" * Deadlines met: {stats.deadlines_met_pct:.1f}%",
130 "",
131 "Details" + (f" ({header})" if header else "") + ":",
132 f" * Incidents in scenario: {stats.total} (handled {stats.handled}, "
133 f"unhandled {len(stats.unhandled)})",
134 f" * On time: {stats.on_time} / {stats.total}",
135 f" * Critical on time: {stats.critical_on_time}",
136 f" * Priority-weighted missed (late or unhandled): {stats.weighted_missed}",
137 " * Per priority (total / handled / on time):",
138 ]
139 for p, ps in stats.per_priority.items():
140 pct = 100.0 * ps.on_time / ps.total
141 lines.append(f" {p:<8} {ps.total} / {ps.handled} / {ps.on_time} ({pct:.1f}% on time)")
142 late = ", ".join(f"{i} (+{d:.1f} s)" for i, d in stats.late) or "none"
143 lines += [
144 f" * Late incidents: {late}",
145 f" * Unhandled incidents: {', '.join(stats.unhandled) or 'none'}",
146 *(f" * {note}" for note in notes),
147 "",
148 "Definitions:",
149 " * Units: seconds (s). Times are absolute simulation times from t = 0.",
150 " * Incidents handled: incidents with an assigned responder.",
151 " * Responders used: distinct responders with at least one assignment.",
152 " * Average response time: mean of (arrival - incident timestamp) over handled incidents only.",
153 " * Deadlines met: on-time arrivals (arrival <= deadline) / ALL incidents;"
154 " unhandled incidents count as missed.",
155 " * Priority-weighted missed: sum of priority weights (Critical 8, High 4, Medium 2, Low 1)"
156 " of late or unhandled incidents.",
157 " * All times are EXPECTED times under the model (Bayesian road status + HMM traffic),"
158 " not observed ground truth.",
159 ]
160 return "\n".join(lines) + "\n"
