1"""@brief Mutation sweep: does the test suite catch typical bugs in EVERY module?
2
3Usage: python tools/mutation_sweep.py (about 25 seconds)
4
5For each mutant (one small, realistic bug) the project is copied to a temporary folder, the bug is
6applied to the copy, and the whole test suite runs there. A mutant is CAUGHT ("killed") if at least
7one test fails; a surviving mutant means a bug of that kind would go unnoticed (a gap in the tests).
8The real source files are never modified, and no bytecode cache is written (no stale .pyc problems).
9Each mutation's text must occur exactly once in its file, otherwise the sweep reports an error
10instead of silently testing nothing.
11"""

130def _copy_project(dest: Path) -> None:
131 """@brief Copy the files needed to run the tests into a temporary folder.
132 @param dest Destination folder.
133 """
134 for name in COPY:
135 src = ROOT / name
136 if src.is_dir():
137 shutil.copytree(src, dest / name, ignore=shutil.ignore_patterns("__pycache__"))
138 else:
139 shutil.copy2(src, dest / name)

142def _run_tests(cwd: Path) -> tuple[bool, str]:
143 """@brief Run the whole test suite in a folder.
144 @param cwd The (temporary) project folder.
145 @return (all tests passed, last line of the pytest output); a timeout counts as caught.
146 """
147 env = {**os.environ, "PYTHONDONTWRITEBYTECODE": "1"}
148 env.pop("RUN_MUTATION_SWEEP", None) # never let the copied suite start another sweep
149 try:
150 proc = subprocess.run([sys.executable, "-m", "pytest", "-x", "-q", "-p", "no:cacheprovider"],
151 cwd=cwd, env=env, capture_output=True, text=True, timeout=600)
152 except subprocess.TimeoutExpired:
153 return False, "timeout (counts as caught)"
154 lines = proc.stdout.strip().splitlines()
155 return proc.returncode == 0, lines[-1] if lines else proc.stderr.strip()[-200:]

158def main() -> int:
159 """@brief Run the sweep and print one line per mutant.
160 @return Exit code 0 if every mutant is caught, 1 otherwise.
161 """
162 with tempfile.TemporaryDirectory(prefix="mutation-sweep-") as tmp:
163 work = Path(tmp)
164 _copy_project(work)
165 ok, summary = _run_tests(work)
166 if not ok:
167 print(f"The UNMUTATED suite fails, sweep aborted: {summary}")
168 return 1
169 print(f"Unmutated suite: {summary}\n")
170 caught, problems = 0, []
171 width = max(len(m.bug) for m in MUTANTS)
172 for k, m in enumerate(MUTANTS, 1):
173 target = work / m.file
174 original = target.read_text()
175 if original.count(m.old) != 1:
176 problems.append(m.bug)
177 print(f"{k:>2}. {m.bug:<{width}} ERROR: mutation text found "
178 f"{original.count(m.old)} times in {m.file}")
179 continue
180 target.write_text(original.replace(m.old, m.new))
181 passed, last = _run_tests(work)
182 target.write_text(original) # restore before the next mutant
183 if passed:
184 problems.append(m.bug)
185 else:
186 caught += 1
187 print(f"{k:>2}. {m.bug:<{width}} {'SURVIVED' if passed else 'caught'} ({last})")
188 print(f"\n{caught} of {len(MUTANTS)} mutants caught.")
189 if problems:
190 print("Not caught / not applied: " + "; ".join(problems))
191 return 0 if not problems else 1
192
193
194if __name__ == "__main__":
195 sys.exit(main())
