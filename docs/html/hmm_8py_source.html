1"""@brief Phase 3: HMM traffic model (Notes 8, forward algorithm / filtering).
2
3Per directed road: X_1 -> X_2 -> ... -> X_n (hidden true traffic level Low/Medium/High), each X_t
4emitting E_t (the observed level).
5
6Forward algorithm, for every observation:
7 1. time elapse : B'(x_t) = Σ_{x_t-1} P(x_t | x_t-1) · B(x_t-1) (skipped for t = 1)
8 2. observe : B(x_t) ∝ P(e_t | x_t) · B'(x_t), then normalise
9
10Convention: model.initial = P(X_1), the state that emits observation 1 (no transition before it).
11Normalising every step keeps the numbers from underflowing; the normalising constants are
12z_t = P(e_t | e_1..e_t-1), so their product is the likelihood P(e_1..e_n) (chain rule).
13We use forward (filtering), not Viterbi: we need the distribution of the CURRENT state, not the
14single most likely history. Works with float and with fractions.Fraction (only +, *, / are used).
15"""

27def predict(model: TrafficModel, belief: Belief, steps: int = 1) -> Belief:
28 """@brief Time elapse only: B'(y) = Σ_x B(x) · P(y | x), repeated `steps` times.
29 @param model The HMM tables.
30 @param belief Current belief over the traffic level.
31 @param steps Number of time steps to predict.
32 @return The predicted belief.
33 """
34 for _ in range(steps):
35 belief = {
36 y: sum(belief[x] * model.transition[x][y] for x in belief)
37 for y in model.initial
38 }
39 return belief

42def observe(model: TrafficModel, belief: Belief, evidence: str) -> tuple[Belief, float]:
43 """@brief Observation step: weight the belief by P(e | x) and normalise.
44 @param model The HMM tables.
45 @param belief Belief after the time-elapse step.
46 @param evidence The observed traffic level.
47 @return (new belief, normaliser z = P(e_t | e_1..e_t-1)).
48 @exception ValueError If the observation has probability 0 under the belief.
49 """
50 weighted = {x: belief[x] * model.emission[x][evidence] for x in belief}
51 z = sum(weighted.values()) # z = P(e_t | e_1..e_{t-1})
52 if z == 0:
53 raise ValueError(f"observation {evidence!r} has probability 0 under the current belief")
54 return {x: p / z for x, p in weighted.items()}, z

57def forward(model: TrafficModel, observations: Sequence[str]) -> tuple[Belief, float]:
58 """@brief Forward algorithm (filtering) over one observation sequence.
59 @param model The HMM tables.
60 @param observations Observed levels in time order.
61 @return (P(X_n | e_1..e_n), likelihood P(e_1..e_n)).
62 @note Checked exactly with fractions on the practicum umbrella example: P(Rain_3 | +,+,-) = 2/5,
63 likelihood 15/128.
64 """
65 belief: Belief = dict(model.initial) # P(X_1): no transition before observation 1
66 likelihood = 1 # int 1 keeps Fraction exact; becomes float with float tables
67 for t, evidence in enumerate(observations):
68 if t > 0:
69 belief = predict(model, belief) # 1. time elapse X_{t-1} -> X_t
70 belief, z = observe(model, belief, evidence) # 2. observe + normalise
71 likelihood *= z
72 return belief, likelihood

75def traffic_beliefs(scenario: Scenario) -> dict[Edge, Belief]:
76 """@brief Traffic "now" (t = 0) for every road with observations (design decision 3).
77
78 Filtered belief after the last observation plus ONE prediction step. Roads without observations
79 are not returned; the cost model gives them the unobserved-traffic multiplier (default 1.0).
80 @param scenario The loaded scenario.
81 @return (from, to) -> belief over Low / Medium / High.
82 """
83 result: dict[Edge, Belief] = {}
84 for edge, observations in scenario.traffic_obs.items():
85 filtered, _ = forward(scenario.traffic_model, observations)
86 result[edge] = predict(scenario.traffic_model, filtered, steps=1)
87 return result

90def expected_multiplier(
91 belief: Belief, multipliers: dict[str, float] = DEFAULT_MULTIPLIERS
92) -> float:
93 """@brief Expected traffic multiplier E[mult] = Σ_x P(x) · mult(x) (design decision 4).
94 @param belief Belief over the traffic level.
95 @param multipliers Multiplier per level (default Low 1.0, Medium 1.25, High 1.5).
96 @return The expected multiplier (between 1.0 and 1.5 with the defaults).

101def stationary_distribution(model: TrafficModel, tol: float = 1e-12, max_steps: int = 100_000) -> Belief:
102 """@brief Stationary distribution π with π = π · T.
103
104 Found by predicting from the initial belief until the belief stops changing. Predicting far
105 ahead converges to π, so far-ahead beliefs lose their road-specific information.
106 @param model The HMM tables.
107 @param tol Convergence tolerance (largest change per step).
108 @param max_steps Upper bound on the number of steps.
109 @return π (small and large: (45, 49, 33)/127; medium has its own tables).
110 @exception RuntimeError If it does not converge within max_steps.
111 """
112 belief: Belief = dict(model.initial)
113 for _ in range(max_steps):
114 nxt = predict(model, belief)
115 if max(abs(nxt[x] - belief[x]) for x in belief) < tol:
116 return nxt
117 belief = nxt
118 raise RuntimeError("prediction did not converge")
