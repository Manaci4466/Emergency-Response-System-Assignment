25def plan_entries(result: SimulationResult) -> list[dict]:
26 """@brief The plan.json entries: EXACTLY the four keys from the assignment.
27
28 time = when the responder starts the assignment; entries are sorted by time, then responder id.
29 @param result The simulation result.
30 @return List of dicts ready for json.dumps.
31 """
32 ordered = sorted(result.assignments, key=lambda a: (a.start, id_key(a.responder_id)))
33 return [
34 {"responder_id": a.responder_id, "incident_id": a.incident_id,
35 "path": list(a.path), "time": a.start}
36 for a in ordered
37 ]

40def _tau(text: str) -> float:
41 """@brief argparse type for --tau: must lie in (0, 1].
42
43 tau = 0 would close every road; tau above 1 is meaningless.
44 @param text The command-line value.
45 @return The value as float.
46 @exception argparse.ArgumentTypeError If out of range.
47 """
48 value = float(text)
49 if not 0 < value <= 1:
50 raise argparse.ArgumentTypeError(f"tau must be in (0, 1], got {value}")
51 return value

54def _multiplier(text: str) -> float:
55 """@brief argparse type for --unobserved-traffic: must be ≥ 1.
56
57 A multiplier below 1 would make a road faster than its free-flow time, which breaks the premise
58 of the A* heuristic (cost ≥ distance / max_speed).
59 @param text The command-line value.
60 @return The value as float.
61 @exception argparse.ArgumentTypeError If below 1.
62 """
63 value = float(text)
64 if value < 1:
65 raise argparse.ArgumentTypeError(f"--unobserved-traffic must be >= 1, got {value}")
66 return value

81def main(argv: list[str] | None = None) -> int:
82 """@brief Run one scenario: load, simulate, write the outputs, validate.
83 @param argv Command-line arguments (default: sys.argv).
84 @return Exit code: 0 if the written plan is valid, 1 if the validator finds errors.
85 """
86 parser = argparse.ArgumentParser(description="Emergency response decision-support system")
87 parser.add_argument("scenario", type=Path, help="scenario folder, e.g. Scenarios/small")
88 parser.add_argument("--out", type=Path, default=None, help="output folder (default output/<name>)")
89 parser.add_argument("--tau", type=_tau, default=0.5,
90 help="road unusable if P(Blocked) >= tau, in (0, 1]")
91 parser.add_argument("--policy", choices=sorted(POLICIES), default="csp")
92 parser.add_argument("--unobserved-traffic", type=_multiplier, default=1.0,
93 help="traffic multiplier for roads without observations, >= 1")
94 parser.add_argument("--node-limit", type=_positive_int, default=NODE_LIMIT,
95 help="CSP search nodes per round before the best plan so far is used")
96 args = parser.parse_args(argv)
97
98 scenario = load_scenario(args.scenario)
99 out = args.out or Path("output") / args.scenario.name
100 out.mkdir(parents=True, exist_ok=True)
101
102 # The A* heuristic needs every road to be at least as long as the straight line: check the data.
103 notes = [f"Route search: {heuristic_description(scenario)}"]
104 if not heuristic_is_admissible(scenario):
105 print(f"WARNING: {heuristic_description(scenario)}; using uniform-cost search")
106
107 result = simulate(scenario, tau=args.tau, unobserved_traffic=args.unobserved_traffic,
108 policy=args.policy, node_limit=args.node_limit)
109 if result.limit_rounds:
110 rounds = ", ".join(f"t = {t:g}" for t in result.limit_rounds)
111 warning = (f"CSP node limit ({args.node_limit}) hit in round(s) {rounds}: "
112 f"best plan found so far used, may not be optimal")
113 print(f"WARNING: {warning}")
114 notes.append(f"WARNING: {warning}")
115 stats = compute_statistics(scenario.incidents, result.assignments, result.unhandled)
116 header = f"policy {args.policy}, tau {args.tau}, unobserved traffic {args.unobserved_traffic}"
117
118 (out / "plan.json").write_text(json.dumps(plan_entries(result), indent=2) + "\n")
119 (out / "statistics.txt").write_text(format_statistics(stats, header, notes))
120
121 avg = "n/a" if stats.avg_response_time is None else f"{stats.avg_response_time:.1f} s"
122 print(f"{args.scenario.name} ({header}): {stats.handled}/{stats.total} handled, "
123 f"{stats.responders_used} responders, avg response {avg}, "
124 f"deadlines met {stats.deadlines_met_pct:.1f}% -> {out}")
125
126 # Validate what was WRITTEN, read back from disk, with the independent validator.
127 plan = json.loads((out / "plan.json").read_text())
128 errors, warnings = validate(plan, scenario, tau=args.tau,
129 unobserved_traffic=args.unobserved_traffic)
130 if not errors: # the statistics check assumes a schema-valid plan
131 errors += check_statistics(plan, scenario, (out / "statistics.txt").read_text(),
132 tau=args.tau, unobserved_traffic=args.unobserved_traffic)
133 for issue in errors:
134 print(f" ERROR {issue.code}: {issue.message}")
135 print(f"plan valid: {len(errors)} errors, {len(warnings)} warnings")
136 return 1 if errors else 0
137
138
139if __name__ == "__main__":
140 sys.exit(main())
