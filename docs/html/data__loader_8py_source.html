1"""@brief Phase 1: load one scenario folder into typed, validated dataclasses.
2
3No AI happens here; this is the problem formulation that the later phases use:
4 - graph (nodes + DIRECTED roads) -> search (A*), Notes 1
5 - incidents (variables) / responders (domains) -> CSP allocation, Notes 2
6 - road_probabilities.json (CPTs) -> Bayes net, Notes 6
7 - traffic_probabilities.json (HMM tables) -> forward algorithm, Notes 8

160 - roads: (from, to) -> Road (directed).

177 def __post_init__(self) -> None:
178 """@brief Build the adjacency list out_edges from the roads."""
179 for node_id in self.nodes:
180 self.out_edges[node_id] = []
181 for road in self.roads.values():
182 self.out_edges[road.source].append(road)

187def _read_csv(path: Path) -> list[dict[str, str]]:
188 """@brief Read a CSV file into a list of dicts, skipping blank lines.
189
190 The scenario files use blank lines as separators. csv.DictReader already skips completely empty
191 lines; rows that contain only commas or spaces are dropped as well. Keys and values are stripped.
192 @param path CSV file.
193 @return One dict per data row.
194 """
195 with path.open(newline="", encoding="utf-8") as f:
196 rows = list(csv.DictReader(f))
197 # DictReader already skips completely empty lines; also drop rows that are only commas/spaces.
198 return [
199 {k.strip(): (v or "").strip() for k, v in row.items()}
200 for row in rows
201 if any((v or "").strip() for v in row.values())
202 ]

214def _check_distribution(dist: dict[str, float], what: str) -> None:
215 """@brief Check that a probability distribution sums to 1.
216 @param dist Value -> probability.
217 @param what Name used in the error message.
218 @exception ValueError If the sum differs from 1 by more than 1e-6.
219 """
220 total = sum(dist.values())
221 if abs(total - 1.0) > 1e-6:
222 raise ValueError(f"{what} sums to {total}, expected 1")
223

225def _check_cpt(cpt: CPT, what: str) -> None:
226 """@brief Check that every row of a conditional probability table sums to 1.
227 @param cpt Table given -> value -> probability.
228 @param what Name used in the error message.
229 @exception ValueError If a row does not sum to 1.
230 """
231 for given, row in cpt.items():
232 _check_distribution(row, f"{what}[{given}]")

237def load_graph(path: Path) -> tuple[dict[str, Node], dict[Edge, Road]]:
238 """@brief Load city_graph.json.
239 @param path Path to city_graph.json.
240 @return (nodes by id, roads by (from, to)).
241 @exception ValueError For an unknown node, a non-positive speed or a duplicate road.
242 """
243 data = _read_json(path)
244 nodes = {
245 n["id"]: Node(n["id"], n["name"], float(n["x"]), float(n["y"]), n["type"])
246 for n in data["nodes"]
247 }
248 roads: dict[Edge, Road] = {}
249 for r in data["roads"]:
250 road = Road(r["from"], r["to"], float(r["distance"]), r["road_type"], float(r["speed"]))
251 if road.source not in nodes or road.target not in nodes:
252 raise ValueError(f"road {road.source}->{road.target} uses an unknown node")
253 if road.speed <= 0:
254 raise ValueError(f"road {road.source}->{road.target} has speed {road.speed}")
255 edge = (road.source, road.target)
256 if edge in roads:
257 raise ValueError(f"duplicate road {edge}")
258 roads[edge] = road
259 return nodes, roads

304def load_reports(path: Path) -> dict[Edge, list[RoadReport]]:
305 """@brief Load road_reports.csv, grouped per DIRECTED road.
306
307 Weather is ONE evidence variable per road in the Bayes net (W -> S -> reports), so all reports for
308 the same directed road must agree on it.
309 @param path Path to road_reports.csv.
310 @return (from, to) -> list of reports.
311 @exception ValueError For an unknown source or inconsistent weather on one road.
312 """
313 reports: dict[Edge, list[RoadReport]] = {}
314 for row in _read_csv(path):
315 if row["source"] not in REPORT_SOURCES:
316 raise ValueError(f"unknown report source {row['source']!r}")
317 edge = (row["from"], row["to"])
318 reports.setdefault(edge, []).append(
319 RoadReport(row["weather"], row["source"], row["reported_status"])
320 )
321 # Weather is ONE evidence variable per road in the Bayes net (W -> S -> reports),
322 # so all reports for the same directed road must agree on it.
323 for edge, road_reports in reports.items():
324 weathers = {r.weather for r in road_reports}
325 if len(weathers) > 1:
326 raise ValueError(f"reports for road {edge} disagree on weather: {sorted(weathers)}")
327 return reports

330def load_traffic_observations(path: Path) -> dict[Edge, list[str]]:
331 """@brief Load observed_traffic.csv: one observation sequence PER DIRECTED ROAD.
332
333 The sequence is sorted by observation_nr, which must be exactly 1..n (a gap would make the HMM
334 treat two distant observations as consecutive time steps).
335 @param path Path to observed_traffic.csv.
336 @return (from, to) -> observed levels in time order.
337 @exception ValueError If the numbers of a road are not 1..n.
338 """
339 numbered: dict[Edge, list[tuple[int, str]]] = {}
340 for row in _read_csv(path):
341 edge = (row["from"], row["to"])
342 numbered.setdefault(edge, []).append((int(row["observation_nr"]), row["traffic_observation"]))
343 result: dict[Edge, list[str]] = {}
344 for edge, seq in numbered.items():
345 seq.sort()
346 numbers = [nr for nr, _ in seq]
347 if numbers != list(range(1, len(seq) + 1)):
348 raise ValueError(f"traffic observations for {edge} are not numbered 1..n: {numbers}")
349 result[edge] = [obs for _, obs in seq]
350 return result

353def load_road_model(path: Path) -> RoadModel:
354 """@brief Load road_probabilities.json into a RoadModel and check every row sums to 1.
355
356 The report tables are mapped from the CSV source names (Citizen, Police, Sensor) to their JSON keys.
357 @param path Path to road_probabilities.json.
358 @return The Bayes-net tables.
359 """
360 data = _read_json(path)
361 model = RoadModel(
362 weather_prior=data["weather_prior"],
363 status_given_weather=data["road_status_given_weather"],
364 report_given_status={src: data[key] for src, key in REPORT_SOURCES.items()},
365 )
366 _check_distribution(model.weather_prior, "weather_prior")
367 _check_cpt(model.status_given_weather, "road_status_given_weather")
368 for src, cpt in model.report_given_status.items():
369 _check_cpt(cpt, f"{src} report")
370 return model

386def load_scenario(folder: str | Path) -> Scenario:
387 """@brief Load and validate a complete scenario folder.
388
389 Besides the per-file checks, it checks cross-file consistency (every responder, incident, report
390 and observation refers to an existing node or road) and fails fast on unknown values such as a
391 misspelled status "Blockd".
392 @param folder Scenario folder, e.g. Scenarios/small.
393 @return The validated Scenario.
394 @exception ValueError For any inconsistency in the data.
395 """
396 folder = Path(folder)
397 nodes, roads = load_graph(folder / "city_graph.json")
398 scenario = Scenario(
399 nodes=nodes,
400 roads=roads,
401 responders=load_responders(folder / "responders.csv"),
402 incidents=load_incidents(folder / "incidents.csv"),
403 reports=load_reports(folder / "road_reports.csv"),
404 traffic_obs=load_traffic_observations(folder / "observed_traffic.csv"),
405 road_model=load_road_model(folder / "road_probabilities.json"),
406 traffic_model=load_traffic_model(folder / "traffic_probabilities.json"),
407 )
408 # Cross-file consistency: everything must refer to existing nodes / roads.
409 for r in scenario.responders:
410 if r.start_location not in nodes:
411 raise ValueError(f"responder {r.id} starts at unknown node {r.start_location}")
412 for i in scenario.incidents:
413 if i.location not in nodes:
414 raise ValueError(f"incident {i.id} at unknown node {i.location}")
415 for edge in list(scenario.reports) + list(scenario.traffic_obs):
416 if edge not in roads:
417 raise ValueError(f"data refers to non-existent road {edge}")
418 # Fail fast on typos ("Blockd"): every value must exist in the matching probability table,
419 # otherwise it would only crash later as a KeyError deep inside bayes.py / hmm.py.
420 weathers = set(scenario.road_model.weather_prior)
421 statuses = set(next(iter(scenario.road_model.status_given_weather.values())))
422 traffic_levels = set(scenario.traffic_model.initial)
423 for edge, road_reports in scenario.reports.items():
424 for rep in road_reports:
425 if rep.weather not in weathers:
426 raise ValueError(f"report on {edge}: unknown weather {rep.weather!r}")
427 if rep.reported_status not in statuses:
428 raise ValueError(f"report on {edge}: unknown status {rep.reported_status!r}")
429 for edge, seq in scenario.traffic_obs.items():
430 for obs in seq:
431 if obs not in traffic_levels:
432 raise ValueError(f"traffic observation on {edge}: unknown level {obs!r}")
433 return scenario
