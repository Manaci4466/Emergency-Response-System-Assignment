63def rounds_by_time(incidents: list[Incident]) -> list[tuple[float, list[Incident]]]:
64 """@brief Group incidents by timestamp, in ascending time order (ids in natural order).
65 @param incidents All incidents of the scenario.
66 @return List of (time, incidents of that time).
67 """
68 groups: dict[float, list[Incident]] = defaultdict(list)
69 for inc in incidents:
70 groups[inc.timestamp].append(inc)
71 return [(t, sorted(groups[t], key=lambda i: id_key(i.id))) for t in sorted(groups)]

74def simulate(
75 scenario: Scenario, tau: float = 0.5, unobserved_traffic: float = 1.0, policy: str = "csp",
76 unreported: str = "prior", node_limit: int = NODE_LIMIT,
77) -> SimulationResult:
78 """@brief Run the whole scenario round by round.
79 @param scenario The scenario.
80 @param tau Blocked threshold.
81 @param unobserved_traffic Multiplier for roads without traffic observations.
82 @param policy "csp" or "greedy".
83 @param unreported "prior" or "open" (see cost_model.edge_costs).
84 @param node_limit CSP node limit per round.
85 @return The SimulationResult.
86 @exception ValueError For an unknown policy.
87 """
88 if policy not in POLICIES:
89 raise ValueError(f"unknown policy {policy!r}, choose from {sorted(POLICIES)}")
90 solve = POLICIES[policy]
91 costs = edge_costs(scenario, tau=tau, unobserved_traffic=unobserved_traffic,
92 unreported=unreported) # once, t = 0
93 states = initial_states(scenario)
94 assignments: list[Assignment] = []
95 unhandled: list[str] = []
96 rounds: list[RoundStats] = []
97 for t, incidents in rounds_by_time(scenario.incidents):
98 # Only THIS round's incidents and the CURRENT states are passed: no look-ahead.
99 if policy == "csp":
100 new, missed, states, st = allocate(scenario, costs, t, incidents, states, node_limit)
101 else:
102 new, missed, states, st = solve(scenario, costs, t, incidents, states)
103 assignments.extend(new)
104 unhandled.extend(missed)
105 rounds.append(RoundStats(t, tuple(i.id for i in incidents), st.nodes, st.hit_limit,
106 st.late_weight, st.response_time))
107 return SimulationResult(policy, tau, unobserved_traffic, assignments, unhandled, rounds)
