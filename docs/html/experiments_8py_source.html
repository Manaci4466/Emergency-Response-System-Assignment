1"""@brief Phase 8: experiments. Run with `python -m src.experiments`; writes results/experiments.md.
2
31-4: sensitivity of the design decisions (tau, unreported roads, unobserved traffic, policy).
4
55-6: Monte Carlo robustness (Notes 6, sampling). The plan is made with EXPECTED costs; here we
6sample N "true" worlds and replay the FIXED plan in each:
7 - each road's true status ~ its Bayes posterior (unreported roads: the prior marginal),
8 - each observed road's true traffic level ~ its HMM belief at t = 0 (unobserved roads: Low),
9 - a job starts at max(its planned time, the responder's finish time in this world),
10 - a route that contains a road that is Blocked in this world fails: the incident is missed, and
11 the responder is stranded, so its later jobs (whose fixed routes start at the incident it never
12 reached) are missed too.
13
14No re-planning: this measures the plan, not the planner.
15"""

93def tau_changes(scenario: Scenario, tau: float, reference: float = 0.5) -> str:
94 """@brief Which roads open or close compared with a reference tau, grouped by P(Blocked).
95 @param scenario The scenario.
96 @param tau The tau to compare.
97 @param reference The reference tau (default 0.5).
98 @return A one-line explanation, e.g. "opens N1→N2, N2→N4 (P = 0.527)".
99 """
100 if tau == reference:
101 return "reference"
102 costs = edge_costs(scenario, tau=tau)
103 ref = usable_roads(scenario, reference)
104 changed: dict[float, list[str]] = {}
105 for e, c in costs.items():
106 if c.usable != (e in ref):
107 changed.setdefault(round(c.p_blocked, 3), []).append(f"{e[0]}→{e[1]}")
108 if not changed:
109 return "no road changes"
110 verb = "opens" if tau > reference else "closes"
111 parts = []
112 for p, roads in sorted(changed.items()):
113 listed = ", ".join(sorted(roads)) if len(roads) <= 4 else f"{len(roads)} roads"
114 parts.append(f"{listed} (P = {p})")
115 return f"{verb} " + "; ".join(parts)

120@dataclass(frozen=True)
121class MonteCarloResult:
122 """@brief Result of replaying one plan in N sampled worlds.
123
124 @par Fields
125 - worlds: number of sampled worlds.
126 - p_any_route_blocked: fraction of worlds in which at least one planned route is blocked.

128 - most_missed: the 3 incidents missed most often, with their fraction.
129 - missed: every incident -> fraction of worlds in which it is missed (late or failed).

138 missed: dict[str, float] # every incident -> fraction of worlds in which it is missed
139 route_blocked: dict[str, float] # incident id -> fraction of worlds its planned route is blocked

142def _sample(dist: Dist, rng: random.Random) -> str:
143 """@brief Draw one value from a discrete distribution (inverse CDF, keys in a fixed order).
144 @param dist Value -> probability.
145 @param rng The random generator (seeded, so results are reproducible).
146 @return The sampled value.
147 """
148 u, acc = rng.random(), 0.0
149 for value, p in dist.items():
150 acc += p
151 if u < acc:
152 return value
153 return next(reversed(dist)) # u fell in the rounding gap at the very top

156def _percentile(sorted_vals: list[float], q: float) -> float:
157 """@brief Nearest-rank percentile of a sorted list.
158 @param sorted_vals Values in ascending order.
159 @param q Quantile between 0 and 1.
160 @return The percentile value.
161 """
162 return sorted_vals[round(q * (len(sorted_vals) - 1))]

165def monte_carlo(
166 scenario: Scenario,
167 assignments: list[Assignment],
168 status: dict[Edge, Dist] | None = None,
169 traffic: dict[Edge, Dist] | None = None,
170 n: int = N_WORLDS,
171 seed: int = SEED,
172) -> MonteCarloResult:
173 """@brief Replay a fixed plan in n sampled worlds (Notes 6, sampling).
174 @param scenario The scenario.
175 @param assignments The plan to replay (its routes are fixed).
176 @param status Road status beliefs to sample from (default: the Bayes posteriors).
177 @param traffic Traffic beliefs to sample from (default: the HMM beliefs at t = 0).
178 @param n Number of worlds.
179 @param seed Random seed (same seed -> identical result).
180 @return The MonteCarloResult.
181 """
182 status = road_status_beliefs(scenario) if status is None else status
183 traffic = traffic_beliefs(scenario) if traffic is None else traffic
184 incidents = {i.id: i for i in scenario.incidents}
185 edges_of = {a.incident_id: list(zip(a.path, a.path[1:])) for a in assignments}
186 needed = sorted({e for es in edges_of.values() for e in es})
187 jobs_by_responder: dict[str, list[Assignment]] = {}
188 for a in sorted(assignments, key=lambda a: (a.start, id_key(a.incident_id))):
189 jobs_by_responder.setdefault(a.responder_id, []).append(a)
190
191 rng = random.Random(seed)
192 met: list[float] = []
193 any_blocked = 0
194 missed = Counter()
195 blocked_route = Counter()
196 for _ in range(n):
197 # 1. sample one world (only the roads the plan uses matter)
198 world_status = {e: _sample(status[e], rng) for e in needed}
199 world_level = {e: _sample(traffic[e], rng) if e in traffic else "Low" for e in needed}
200 if any(s == "Blocked" for s in world_status.values()):
201 any_blocked += 1
202 # 2. replay the fixed plan
203 on_time = 0
204 for jobs in jobs_by_responder.values():
205 free_at, stranded = 0.0, False
206 for a in jobs:
207 inc = incidents[a.incident_id]
208 route = edges_of[a.incident_id]
209 if any(world_status[e] == "Blocked" for e in route):
210 blocked_route[a.incident_id] += 1
211 if stranded or any(world_status[e] == "Blocked" for e in route):
212 stranded = True
213 missed[a.incident_id] += 1
214 continue
215 start = max(a.start, free_at)
216 travel = sum(scenario.roads[e].distance / scenario.roads[e].speed
217 * STATUS_MULT[world_status[e]] * DEFAULT_MULTIPLIERS[world_level[e]]
218 for e in route)
219 arrival = start + travel
220 if arrival <= inc.deadline:
221 on_time += 1
222 else:
223 missed[a.incident_id] += 1
224 free_at = arrival + inc.time_required
225 for iid in incidents: # unhandled in the plan: missed in every world
226 if iid not in edges_of:
227 missed[iid] += 1
228 met.append(100.0 * on_time / len(incidents))
229
230 met.sort()
231 top = sorted(missed.items(), key=lambda kv: (-kv[1], id_key(kv[0])))[:3]
232 return MonteCarloResult(
233 worlds=n,
234 p_any_route_blocked=any_blocked / n,
235 mean_deadlines_met=sum(met) / n,
236 p5_deadlines_met=_percentile(met, 0.05),
237 p95_deadlines_met=_percentile(met, 0.95),
238 most_missed=[(i, c / n) for i, c in top],
239 missed={i: missed[i] / n for i in incidents},
240 route_blocked={i: blocked_route[i] / n for i in edges_of},
241 )

252def unobserved_traffic_rows(scen: dict[str, Scenario]) -> list[tuple[str, float, Metrics]]:
253 """@brief Experiment 3: each scenario with unobserved traffic ×1.0 and ×its own stationary multiplier.
254 @param scen Scenario name -> scenario.
255 @return List of (name, multiplier, metrics).
256 """
257 rows = []
258 for n, s in scen.items():
259 for mult in [1.0, stationary_multiplier(s)]:
260 rows.append((n, mult, metrics(s, simulate(s, unobserved_traffic=mult))))
261 return rows

266def run_all() -> str:
267 """@brief Run all experiments.
268 @return The markdown text of results/experiments.md.
269 """
270 scen = {n: load_scenario(ROOT / "Scenarios" / n) for n in NAMES}
271 out: list[str] = ["# Experiments", "",
272 "Generated by `python -m src.experiments`. All times are EXPECTED seconds under "
273 "the model unless stated otherwise. Weighted missed = Σ priority weight "
274 "(Critical 8, High 4, Medium 2, Low 1) of late or unhandled incidents.", ""]
275
276 out += ["## 1. Threshold tau (CSP)", "",
277 f"| Scenario | tau | Usable roads | {METRIC_HEADER} | Change vs tau 0.5 |",
278 "|---|---|---|---|---|---|---|---|---|"]
279 for n in NAMES:
280 for tau in TAUS:
281 m = metrics(scen[n], simulate(scen[n], tau=tau))
282 out.append(f"| {n} | {tau} | {len(usable_roads(scen[n], tau))}/{len(scen[n].roads)} | "
283 f"{_fmt(m)} | {tau_changes(scen[n], tau)} |")
284
285 out += ["", "## 2. Unreported roads (decision 2), CSP, tau 0.5", "",
286 f"| Scenario | Unreported roads | {METRIC_HEADER} |", "|---|---|---|---|---|---|---|"]
287 for n in NAMES:
288 for mode in ["prior", "open"]:
289 out.append(f"| {n} | {mode} | {_fmt(metrics(scen[n], simulate(scen[n], unreported=mode)))} |")
290
291 out += ["", "## 3. Unobserved traffic (decision 5), CSP, tau 0.5", "",
292 "The alternative multiplier is the expected multiplier under each scenario's OWN "
293 "stationary distribution (medium has different traffic tables).", "",
294 f"| Scenario | Unobserved multiplier | {METRIC_HEADER} |", "|---|---|---|---|---|---|---|"]
295 for n, mult, m in unobserved_traffic_rows(scen):
296 out.append(f"| {n} | {mult:.4f} | {_fmt(m)} |")
297
298 out += ["", "## 4. CSP vs greedy, tau 0.5", "",
299 f"| Scenario | Policy | {METRIC_HEADER} |", "|---|---|---|---|---|---|---|"]
300 for n in NAMES:
301 for policy in ["csp", "greedy"]:
302 out.append(f"| {n} | {policy} | {_fmt(metrics(scen[n], simulate(scen[n], policy=policy)))} |")
303
304 out += ["", f"## 5. Monte Carlo robustness of the CSP plan (tau 0.5, N = {N_WORLDS}, seed {SEED})",
305 "", "| Scenario | Planned deadlines met | P(≥1 planned route blocked) | Mean deadlines met "
306 "| 5th–95th percentile | Most often missed (fraction of worlds) |",
307 "|---|---|---|---|---|---|"]
308 i2_line = ""
309 for n in NAMES:
310 res = simulate(scen[n])
311 planned = metrics(scen[n], res).deadlines_met_pct
312 mc = monte_carlo(scen[n], res.assignments)
313 top = ", ".join(f"{i} ({f:.2f})" for i, f in mc.most_missed)
314 out.append(f"| {n} | {planned:.1f}% | {mc.p_any_route_blocked:.3f} | "
315 f"{mc.mean_deadlines_met:.1f}% | {mc.p5_deadlines_met:.1f}–{mc.p95_deadlines_met:.1f}% "
316 f"| {top} |")
317 if n == "small":
318 a = next(x for x in res.assignments if x.incident_id == "I2")
319 i2_line = (f"Small I2 (Critical): {a.responder_id}'s route {'→'.join(a.path)} is blocked in "
320 f"{mc.route_blocked['I2']:.3f} of the worlds "
321 f"(two unreported roads: 1 − 0.865² = {1 - 0.865 ** 2:.3f}).")
322 out += ["", i2_line, ""]
323
324 out += [f"## 6. Planned vs Monte Carlo per tau (CSP, N = {N_WORLDS}, seed {SEED})", "",
325 "A higher tau makes the PLANNED numbers look better by using roads that are probably "
326 "blocked; the sampled worlds show what that costs.", "",
327 "| Scenario | tau | Planned deadlines met | Monte Carlo mean | 5th–95th percentile "
328 "| P(≥1 planned route blocked) |", "|---|---|---|---|---|---|"]
329 for n in NAMES:
330 for tau in [0.5, 0.6, 0.9]:
331 res = simulate(scen[n], tau=tau)
332 mc = monte_carlo(scen[n], res.assignments)
333 out.append(f"| {n} | {tau} | {metrics(scen[n], res).deadlines_met_pct:.1f}% | "
334 f"{mc.mean_deadlines_met:.1f}% | {mc.p5_deadlines_met:.1f}–"
335 f"{mc.p95_deadlines_met:.1f}% | {mc.p_any_route_blocked:.3f} |")
336 out.append("")
337 return "\n".join(out)
