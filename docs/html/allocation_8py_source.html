1"""@brief Phase 5: responder allocation as a CSP, solved per round (practicum 1: backtracking, MRV,
2forward checking) with branch and bound on a lexicographic objective.
3
4One round = the incidents that become known at time `now` (no clairvoyance).
5 - Variables : this round's incidents.
6 - Domains : responders with the required capability that can reach the incident
7 (unary constraints, filtered first; empty domain -> unhandled).
8 - Hard : capability, reachability, one job at a time (queueing: the next job starts when the
9 previous one finishes, at its location).
10 - Soft : deadlines. Late help is better than no help, so lateness is a PENALTY, never a
11 reason to fail a branch.
12 - Objective : lexicographic (1) Σ priority weight of LATE incidents, (2) Σ (arrival - timestamp).
13
14Because the assignment order also fixes the job order of a shared responder, MRV here affects the
15QUALITY of the solution, not only the speed (in a classical CSP variable order only changes speed).
16Without MRV (plain id order) medium t=0 gets worse: R10 does I2 before I3, and the Critical I3
17arrives late at 41.6 s (late weight 8 instead of 4).
18
19@note Limitation: the jobs of one responder are done in the order the search assigns them, and the
20search order is fixed by MRV, so the CSP is not exhaustive over every job order.
21"""

91def id_key(identifier: str) -> tuple[str, int]:
92 """@brief Natural sort key for ids, so I2 sorts before I10.
93 @param identifier An id such as "I10" or "R2".
94 @return (prefix, number), or (identifier, 0) if it has no number.
95 """
96 m = re.fullmatch(r"(\D*)(\d+)", identifier)
97 return (m.group(1), int(m.group(2))) if m else (identifier, 0)

122 def route(self, a: str, b: str) -> tuple[list[str], float] | None:
123 """@brief Cached A* route from a to b.
124 @param a Start node id.
125 @param b Goal node id.
126 @return (path, travel time in s), or None if unreachable.
127 """
128 if (a, b) not in self._routes:
129 res = astar(self.scenario, self.costs, a, b)
130 self._routes[(a, b)] = None if res is None else (res[0], res[1])
131 return self._routes[(a, b)]

133 def option(self, inc: Incident, rid: str, st: ResponderState) -> Assignment | None:
134 """@brief The assignment that results from sending responder rid (in state st) to an incident.
135
136 start = max(now, free_at) (queueing), arrival = start + A* time, finish = arrival + time_required.
137 @param inc The incident.
138 @param rid The responder id.
139 @param st The responder's current state in this branch.
140 @return The Assignment, or None if the responder cannot reach the incident.
141 """
142 r = self.route(st.location, inc.location)
143 if r is None:
144 return None
145 path, travel = r
146 start = max(self.now, st.free_at) # wait until the responder is free (queueing)
147 arrival = start + travel
148 return Assignment(rid, inc.id, tuple(path), start, arrival,
149 arrival + inc.time_required, arrival > inc.deadline)

179def allocate(
180 scenario: Scenario,
181 costs: dict[Edge, EdgeCost],
182 now: float,
183 incidents: Sequence[Incident],
184 states: dict[str, ResponderState],
185 node_limit: int = NODE_LIMIT,
186 prune: bool = True,
187) -> AllocationResult:
188 """@brief Solve ONE round with backtracking + MRV + forward checking + branch and bound.
189
190 Forward checking recomputes every remaining incident's ON-TIME set after each assignment. An empty
191 set does not fail the branch (deadlines are soft): its weight is added to a lower bound, and the
192 branch is pruned when (bound, response time so far) ≥ the best plan found. The bound is valid
193 because extra jobs only make a responder free later and elsewhere, which by the triangle
194 inequality of shortest paths never makes it arrive earlier.
195 @param scenario The scenario.
196 @param costs Edge costs.
197 @param now Time of the round (s).
198 @param incidents The incidents that become known at `now`.
199 @param states Responder states at the start of the round.
200 @param node_limit Maximum search nodes; if hit, the best plan found so far is used (or greedy if
201 no complete plan was found yet).
202 @param prune False switches branch and bound off (same answer, more nodes); used by tests only.
203 @return (assignments, unhandled incident ids, new responder states, AllocationStats).
204 """
205 planner = _Planner(scenario, costs, now)
206 domains = planner.domains(incidents, states)
207 unhandled = sorted((i.id for i in incidents if not domains[i.id]), key=id_key)
208 variables = [i for i in incidents if domains[i.id]]
209
210 best_obj: Objective = (math.inf, math.inf)
211 best: tuple[list[Assignment], list[str]] = ([], [])
212 nodes = 0
213 hit_limit = False
214
215 def on_time(inc: Incident, sts: dict[str, ResponderState]) -> list[Assignment]:
216 """@brief Assignments of `inc` that arrive on time, given the branch's responder states."""
217 opts = (planner.option(inc, rid, sts[rid]) for rid in domains[inc.id])
218 return [a for a in opts if a is not None and not a.late]

220 def search(
221 unassigned: list[Incident], sts: dict[str, ResponderState],
222 chosen: list[Assignment], skipped: list[str], late_w: float, resp: float,
223 ) -> None:
224 """@brief Recursive backtracking over the unassigned incidents of one branch."""
225 nonlocal best_obj, best, nodes, hit_limit
226 if hit_limit:
227 return
228 if not unassigned: # complete assignment: keep it if strictly better
229 if (late_w, resp) < best_obj:
230 best_obj, best = (late_w, resp), (list(chosen), list(skipped))
231 return
232
233 # Forward checking: recompute every remaining incident's ON-TIME set in this branch.
234 # An empty set does NOT fail the branch (deadlines are soft); it only means that incident
235 # is guaranteed late, so its weight is added to a lower bound. The bound is valid because
236 # assigning more jobs can only make a responder free later and somewhere else, and by the
237 # triangle inequality of shortest paths that never makes it arrive earlier.
238 ontime = {i.id: on_time(i, sts) for i in unassigned}
239 bound = late_w + sum(WEIGHTS[i.priority] for i in unassigned if not ontime[i.id])
240 if prune and (bound, resp) >= best_obj: # branch and bound: cannot beat the best
241 return
242
243 # MRV: fewest on-time responders first (ties: higher weight, earlier deadline, id).
244 inc = min(unassigned, key=lambda i: (len(ontime[i.id]), -WEIGHTS[i.priority],
245 i.deadline, id_key(i.id)))
246 rest = [i for i in unassigned if i is not inc]
247 options = [a for rid in domains[inc.id]
248 if (a := planner.option(inc, rid, sts[rid])) is not None]
249 options.sort(key=lambda a: (a.arrival, id_key(a.responder_id))) # earliest arrival first
250
251 if not options: # nobody can reach it from where they are in THIS branch: counts as missed
252 search(rest, sts, chosen, skipped + [inc.id], late_w + WEIGHTS[inc.priority], resp)
253 return
254 for a in options:
255 nodes += 1
256 if nodes > node_limit:
257 hit_limit = True
258 return
259 search(rest, _apply(sts, a, inc.location), chosen + [a], skipped,
260 late_w + (WEIGHTS[inc.priority] if a.late else 0),
261 resp + (a.arrival - inc.timestamp))
262
263 search(variables, states, [], [], 0.0, 0.0)
264 if best_obj == (math.inf, math.inf):
265 # The node limit was hit before even ONE complete plan was found (only possible when
266 # node_limit < number of incidents): fall back to greedy so no incident silently vanishes.
267 g_assign, g_unhandled, g_states, g_stats = greedy(scenario, costs, now, incidents, states)
268 return g_assign, g_unhandled, g_states, AllocationStats(
269 nodes, True, g_stats.late_weight, g_stats.response_time)
270 assignments, skipped = best
271 new_states = dict(states)
272 by_id = {i.id: i for i in incidents}
273 for a in sorted(assignments, key=lambda a: a.start):
274 new_states[a.responder_id] = ResponderState(by_id[a.incident_id].location, a.finish)
275 stats = AllocationStats(nodes, hit_limit, best_obj[0], best_obj[1])
276 return assignments, sorted(unhandled + skipped, key=id_key), new_states, stats

279def greedy(
280 scenario: Scenario,
281 costs: dict[Edge, EdgeCost],
282 now: float,
283 incidents: Sequence[Incident],
284 states: dict[str, ResponderState],
285) -> AllocationResult:
286 """@brief Baseline: incidents in id order, each gets the capable responder that arrives earliest.
287 @param scenario The scenario.
288 @param costs Edge costs.
289 @param now Time of the round (s).
290 @param incidents The round's incidents.
291 @param states Responder states at the start of the round.
292 @return Same format as allocate (nodes = one decision per incident).
293 """
294 planner = _Planner(scenario, costs, now)
295 sts = dict(states)
296 assignments: list[Assignment] = []
297 unhandled: list[str] = []
298 late_w = resp = 0.0
299 for inc in sorted(incidents, key=lambda i: id_key(i.id)):
300 options = [a for r in scenario.responders if inc.required_capability in r.capabilities
301 if (a := planner.option(inc, r.id, sts[r.id])) is not None]
302 if not options:
303 unhandled.append(inc.id)
304 continue
305 a = min(options, key=lambda a: (a.arrival, id_key(a.responder_id)))
306 assignments.append(a)
307 sts = _apply(sts, a, inc.location)
308 late_w += WEIGHTS[inc.priority] if a.late else 0
309 resp += a.arrival - inc.timestamp
310 # nodes = one decision per incident (greedy never backtracks)
311 return assignments, unhandled, sts, AllocationStats(len(incidents), False, late_w, resp)
