1"""@brief Phase 2: Bayesian road status (Notes 6, inference by enumeration).
2
3Network per directed road: Weather -> Status -> {Citizen, Police, Sensor}
4 - query : Status S (Open / Congested / Blocked)
5 - evidence : the reports, and the weather if it was reported
6 - hidden : the weather when it is NOT reported (summed out with the prior)
7
8S is each report's ONLY parent and there are no edges between the reports, so the reports are
9conditionally independent given S and their likelihoods can simply be multiplied:
10 P(S | w, r1..rn) ∝ P(S | w) · Π_i P(r_i | S)
11P(w) is the same constant for every value of S, so normalisation removes it.
12"""

20def normalise(dist: dict[str, float]) -> dict[str, float]:
21 """@brief Divide by the total so the values sum to 1 (the '∝' step).
22 @param dist Unnormalised non-negative values.
23 @return The normalised distribution.
24 @exception ValueError If all values are zero.
25 """
26 total = sum(dist.values())
27 if total <= 0:
28 raise ValueError(f"cannot normalise, all probabilities are zero: {dist}")
29 return {k: v / total for k, v in dist.items()}

32def status_prior(model: RoadModel, weather: str | None) -> dict[str, float]:
33 """@brief Prior over the road status, before the reports are used.
34
35 If the weather is observed this is the table row P(S | w); otherwise the weather is summed out:
36 P(S) = Σ_w P(w) · P(S | w), which gives Open 0.685, Congested 0.18, Blocked 0.135 (decision 2).
37 @param model The Bayes-net tables.
38 @param weather Observed weather, or None if unknown.
39 @return Status -> probability.
40 """
41 if weather is not None:
42 return dict(model.status_given_weather[weather])
43 statuses = next(iter(model.status_given_weather.values())).keys()
44 return {
45 s: sum(p_w * model.status_given_weather[w][s] for w, p_w in model.weather_prior.items())
46 for s in statuses
47 }

50def status_posterior(
51 model: RoadModel, weather: str | None, reports: Sequence[RoadReport]
52) -> dict[str, float]:
53 """@brief Posterior P(S | weather, reports) ∝ P(S | weather) · Π P(report | S), normalised.
54
55 Any subset of sources may be present: a missing source simply contributes no factor.
56 @param model The Bayes-net tables (report CPTs: outer key = TRUE status).
57 @param weather Observed weather, or None to sum it out with the prior.
58 @param reports The reports for this road.
59 @return Status -> posterior probability.
60 @note Example: small N1->N2 (Rain; Blocked, Blocked, Congested) gives O 0.014, C 0.459, B 0.527.
61 """
62 unnormalised = status_prior(model, weather)
63 for report in reports:
64 cpt = model.report_given_status[report.source] # outer key = TRUE status
65 for s in unnormalised:
66 unnormalised[s] *= cpt[s][report.reported_status]
67 return normalise(unnormalised)

70def road_status_beliefs(scenario: Scenario) -> dict[Edge, dict[str, float]]:
71 """@brief Posterior status for EVERY directed road of the scenario.
72
73 A reported road takes its weather from its reports (the loader guarantees they agree); an
74 unreported road has no weather and no reports and gets the prior marginal P(S).
75 @param scenario The loaded scenario.
76 @return (from, to) -> status distribution.
77 """
78 beliefs: dict[Edge, dict[str, float]] = {}
79 for edge in scenario.roads:
80 reports = scenario.reports.get(edge, [])
81 weather = reports[0].weather if reports else None
82 beliefs[edge] = status_posterior(scenario.road_model, weather, reports)
83 return beliefs
